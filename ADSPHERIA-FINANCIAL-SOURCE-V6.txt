import { supabaseAdmin } from "@/lib/supabase/admin";

export type EscrowAction =
  | "FUND"
  | "HOLD"
  | "RELEASE"
  | "REFUND"
  | "DISPUTE"
  | "AUTO_RELEASE";

const COMMISSION_RATE = 0.1;

export async function escrowEngine(
  action: EscrowAction,
  escrowId: string
) {
  if (!escrowId) {
    throw new Error("Missing escrowId");
  }

  const { data: escrow, error: fetchError } = await supabaseAdmin
    .from("escrow_orders")
    .select("*")
    .eq("id", escrowId)
    .single();

  if (fetchError || !escrow) {
    throw new Error("Escrow not found");
  }

  const now = new Date().toISOString();

  switch (action) {
    case "FUND": {
      if (escrow.status === "funded") {
        return escrow;
      }

      if (escrow.status !== "pending") {
        throw new Error(
          `Cannot fund escrow from status: ${escrow.status}`
        );
      }

      const autoRelease = new Date();
      autoRelease.setDate(autoRelease.getDate() + 7);

      const autoReleaseAt = autoRelease.toISOString();

      const { data, error } = await supabaseAdmin
        .from("escrow_orders")
        .update({
          status: "funded",
          funded_at: now,
          auto_release_at: autoReleaseAt,
          buyer_confirmation_deadline: autoReleaseAt,
          is_locked: true,
          updated_at: now,
        })
        .eq("id", escrowId)
        .eq("status", "pending")
        .select()
        .single();

      if (error) throw error;

      console.log("💰 Escrow FUNDED:", escrowId);

      return data;
    }

    case "HOLD": {
      if (escrow.status === "released") {
        throw new Error("Cannot hold a released escrow");
      }

      const { data, error } = await supabaseAdmin
        .from("escrow_orders")
        .update({
          is_locked: true,
          updated_at: now,
        })
        .eq("id", escrowId)
        .select()
        .single();

      if (error) throw error;

      console.log("⏸ Escrow HELD:", escrowId);

      return data;
    }

    case "RELEASE": {
      if (escrow.status === "released") {
        return escrow;
      }

      if (
        escrow.status !== "funded" &&
        escrow.status !== "delivered" &&
        escrow.status !== "disputed"
      ) {
        throw new Error(
          `Escrow not ready for release from status: ${escrow.status}`
        );
      }

      if (escrow.dispute_status === "open" ||
          escrow.dispute_status === "under_review") {
        throw new Error("Cannot release escrow while dispute is active");
      }

      const commission = Number(escrow.amount) * COMMISSION_RATE;
      const sellerAmount = Number(escrow.amount) - commission;

      const { data, error } = await supabaseAdmin
        .from("escrow_orders")
        .update({
          status: "released",
          released_at: now,
          commission,
          seller_amount: sellerAmount,
          is_locked: false,
          updated_at: now,
        })
        .eq("id", escrowId)
        .in("status", ["funded", "delivered"])
        .select()
        .single();

      if (error) throw error;

      console.log("🏁 Escrow RELEASED:", escrowId);

      return data;
    }

    case "AUTO_RELEASE": {
      if (escrow.status === "released") {
        return escrow;
      }

      if (
        escrow.status !== "funded" &&
        escrow.status !== "delivered"
      ) {
        return escrow;
      }

      if (
        escrow.dispute_status === "open" ||
        escrow.dispute_status === "under_review"
      ) {
        console.log(
          "🛑 AUTO RELEASE BLOCKED — active dispute:",
          escrowId
        );

        return escrow;
      }

      if (
        escrow.auto_release_at &&
        new Date(escrow.auto_release_at) > new Date()
      ) {
        return escrow;
      }

      const commission = Number(escrow.amount) * COMMISSION_RATE;
      const sellerAmount = Number(escrow.amount) - commission;

      const { data, error } = await supabaseAdmin
        .from("escrow_orders")
        .update({
          status: "released",
          released_at: now,
          commission,
          seller_amount: sellerAmount,
          is_auto_released: true,
          is_locked: false,
          updated_at: now,
        })
        .eq("id", escrowId)
        .in("status", ["funded", "delivered"])
        .select()
        .single();

      if (error) throw error;

      console.log("⏱ AUTO RELEASED:", escrowId);

      return data;
    }

    case "REFUND": {
      if (escrow.status === "refunded") {
        return escrow;
      }

      if (escrow.status === "released") {
        throw new Error("Cannot refund a released escrow");
      }

      const { data, error } = await supabaseAdmin
        .from("escrow_orders")
        .update({
          status: "refunded",
          refunded_at: now,
          is_locked: false,
          updated_at: now,
        })
        .eq("id", escrowId)
        .not("status", "eq", "released")
        .select()
        .single();

      if (error) throw error;

      console.log("↩️ Escrow REFUNDED:", escrowId);

      return data;
    }

    case "DISPUTE": {
      if (escrow.status === "released") {
        throw new Error("Cannot dispute a released escrow");
      }

      const { data, error } = await supabaseAdmin
        .from("escrow_orders")
        .update({
          status: "disputed",
          dispute_status: "open",
          disputed_at: now,
          is_locked: true,
          updated_at: now,
        })
        .eq("id", escrowId)
        .not("status", "eq", "released")
        .select()
        .single();

      if (error) throw error;

      console.log("⚠️ Escrow DISPUTED:", escrowId);

      return data;
    }

    default:
      throw new Error(`Unknown escrow action: ${action}`);
  }
}

/* =========================
   AUTO RELEASE ENGINE v5
   SAFE + IDPOTENT + DISPUTE SAFE
========================= */

export async function autoReleaseEngine() {
  const supabase = await createServerSupabase();

  const now = new Date().toISOString();

  // 1. Get ONLY eligible escrows
  const { data: escrows, error } = await supabase
    .from("escrow_orders")
    .select("*")
    .eq("status", "delivered")
    .lte("auto_release_at", now);

  if (error) {
    console.error("Auto release fetch error:", error.message);
    return { processed: 0, released: 0 };
  }

  if (!escrows || escrows.length === 0) {
    return {
      processed: 0,
      released: 0,
    };
  }

  let released = 0;

  // 2. Process each escrow safely
  for (const escrow of escrows) {
    try {
      // 🚫 BLOCK DISPUTES
      if (escrow.status === "disputed") continue;

      // 🚫 DOUBLE CHECK (extra safety layer)
      const { data: fresh } = await supabase
        .from("escrow_orders")
        .select("status")
        .eq("id", escrow.id)
        .single();

      if (!fresh) continue;

      if (fresh.status !== "delivered") {
        continue;
      }

      // 3. SAFE UPDATE (IDEMPOTENT)
      const { error: updateError } = await supabase
        .from("escrow_orders")
        .update({
          status: "released",
          released_at: now,
        })
        .eq("id", escrow.id)
        .eq("status", "delivered"); // prevents double release

      if (updateError) {
        console.error("Release failed:", escrow.id, updateError.message);
        continue;
      }

      released++;

      console.log("💰 AUTO RELEASED:", escrow.id);
    } catch (err) {
      console.error("Auto release crash:", err);
    }
  }

  return {
    processed: escrows.length,
    released,
  };
}

import { createServerSupabase } from "@/lib/supabase/server";

type Action = "refund" | "release";

export async function resolveDispute({
  disputeId,
  action,
  adminId,
}: {
  disputeId: string;
  action: Action;
  adminId: string;
}) {
  const supabase = await createServerSupabase();

  // 1. Get dispute
  const { data: dispute, error } = await supabase
    .from("disputes")
    .select("*")
    .eq("id", disputeId)
    .single();

  if (error || !dispute) {
    throw new Error("Dispute not found");
  }

  // 2. Safety check (must still be open)
  if (dispute.status !== "open") {
    throw new Error("Dispute already resolved");
  }

  // 3. Get escrow
  const { data: escrow } = await supabase
    .from("escrow_orders")
    .select("*")
    .eq("id", dispute.escrow_id)
    .single();

  if (!escrow) {
    throw new Error("Escrow not found");
  }

  // 4. Lock escrow during resolution
  await supabase
    .from("escrow_orders")
    .update({
      is_locked: true,
      status: "disputed",
    })
    .eq("id", escrow.id);

  // 5. Apply decision
  if (action === "release") {
    await supabase
      .from("escrow_orders")
      .update({
        status: "released",
        is_locked: false,
        updated_at: new Date().toISOString(),
      })
      .eq("id", escrow.id);
  }

  if (action === "refund") {
    await supabase
      .from("escrow_orders")
      .update({
        status: "refunded",
        is_locked: false,
        updated_at: new Date().toISOString(),
      })
      .eq("id", escrow.id);
  }

  // 6. Close dispute
  await supabase
    .from("disputes")
    .update({
      status: "resolved",
      resolution: action,
      resolved_by: adminId,
      updated_at: new Date().toISOString(),
    })
    .eq("id", disputeId);

  return {
    success: true,
    action,
    escrowId: escrow.id,
  };
}

import { createServerSupabase } from "@/lib/supabase/server";

export type AgreementAction =
  | "CREATE"
  | "ACCEPT"
  | "REJECT"
  | "COMPLETE"
  | "CANCEL";

/* =========================
   AGREEMENT ENGINE
========================= */
export async function agreementEngine(
  action: AgreementAction,
  agreementId: string
) {
  const supabase = await createServerSupabase();

  // 1. Fetch agreement
  const { data: agreement, error } = await supabase
    .from("agreements")
    .select("*")
    .eq("id", agreementId)
    .single();

  if (error || !agreement) {
    throw new Error("Agreement not found");
  }

  /* =========================
     ACCEPT → CREATE ESCROW
  ========================= */
  if (action === "ACCEPT") {
    if (agreement.status !== "pending") {
      throw new Error("Already processed");
    }

    // update agreement
    const { data: updated, error: updateError } = await supabase
      .from("agreements")
      .update({
        status: "active",
        accepted_at: new Date().toISOString(),
      })
      .eq("id", agreementId)
      .select()
      .single();

    if (updateError) throw updateError;

    // prevent duplicate escrow
    const { data: existing } = await supabase
      .from("escrow_orders")
      .select("id")
      .eq("agreement_id", agreementId)
      .maybeSingle();

    if (existing) {
      return { agreement: updated, escrow: existing };
    }

    // create escrow
    const { data: escrow, error: escrowError } = await supabase
      .from("escrow_orders")
      .insert({
        agreement_id: agreementId,
        conversation_id: agreement.conversation_id,
        buyer_id: agreement.buyer_id,
        seller_id: agreement.seller_id,
        amount: agreement.price,
        currency: agreement.currency ?? "usd",
        status: "pending",
        is_locked: true,
        auto_release_at: new Date(
          Date.now() + 7 * 24 * 60 * 60 * 1000
        ).toISOString(),
      })
      .select()
      .single();

    if (escrowError) throw escrowError;

    return { agreement: updated, escrow };
  }

  /* =========================
     REJECT
  ========================= */
  if (action === "REJECT") {
    const { data, error: rejectError } = await supabase
      .from("agreements")
      .update({
        status: "rejected",
      })
      .eq("id", agreementId)
      .select()
      .single();

    if (rejectError) throw rejectError;

    return data;
  }

  /* =========================
     COMPLETE
  ========================= */
  if (action === "COMPLETE") {
    const { data, error: completeError } = await supabase
      .from("agreements")
      .update({
        status: "completed",
        completed_at: new Date().toISOString(),
      })
      .eq("id", agreementId)
      .select()
      .single();

    if (completeError) throw completeError;

    return data;
  }

  /* =========================
     CREATE (FUTURE USE)
  ========================= */
  if (action === "CREATE") {
    throw new Error("Create flow not implemented yet");
  }

  /* =========================
     CANCEL
  ========================= */
  if (action === "CANCEL") {
    const { data, error: cancelError } = await supabase
      .from("agreements")
      .update({
        status: "cancelled",
      })
      .eq("id", agreementId)
      .select()
      .single();

    if (cancelError) throw cancelError;

    return data;
  }

  throw new Error("Invalid action");
}

import { createServerSupabase } from "@/lib/supabase/server";
import { escrowEngine } from "@/lib/domain/escrow/escrowEngine";

export const cronService = {
  async expireBoosts() {
    const supabase = await createServerSupabase();

    const { data, error } = await supabase.rpc(
      "expire_expired_boosts"
    );

    if (error) {
      throw new Error(
        `Boost expiration failed: ${error.message}`
      );
    }

    return {
      processed: Number(data?.processed ?? 0),
      expired: Number(data?.expired ?? 0),
    };
  },

  async autoReleaseEscrows() {
    const supabase = await createServerSupabase();
    const now = new Date().toISOString();

    const { data: escrows, error } = await supabase
      .from("escrow_orders")
      .select("id")
      .eq("status", "delivered")
      .lte("auto_release_at", now);

    if (error) {
      throw new Error(
        `Auto-release fetch failed: ${error.message}`
      );
    }

    let released = 0;
    let failed = 0;

    for (const escrow of escrows ?? []) {
      try {
        await escrowEngine("AUTO_RELEASE", escrow.id);
        released++;
      } catch (error) {
        failed++;

        console.error(
          "Auto-release failed:",
          escrow.id,
          error instanceof Error ? error.message : error
        );
      }
    }

    return {
      processed: escrows?.length ?? 0,
      released,
      failed,
    };
  },
};

      await chapaResponse.json();

        {
          success:false,

        {
          status:500,
        }
      );

    return NextResponse.json({

      success:true,

      checkout_url:
        chapaData.data.checkout_url,

      tx_ref,

  } catch(error:any){


    console.error(
      "CHAPA INITIALIZE ERROR:",
      error
    );


    return NextResponse.json(
      {
        success:false,

        message:
          error.message ||
          "Server error",
      },
      {
        status:500,
      }
    );

  }

}

    if (!signature) {
      return NextResponse.json(
        { error: "Missing stripe signature" },
        { status: 400 }
      );
    }

    /* =========================
       VERIFY EVENT
    ========================= */
    let event: Stripe.Event;

    try {
      event = stripe.webhooks.constructEvent(
        body,
        signature,
        process.env.STRIPE_WEBHOOK_SECRET!
      );
    } catch (err) {
      return NextResponse.json(
        { error: "Invalid signature" },
        { status: 400 }
      );
    }

    const eventId = event.id;

    /* =========================
       IDEMPOTENCY CHECK
    ========================= */
    const { data: existing } = await supabaseAdmin
      .from("stripe_events")
      .select("id")
      .eq("id", eventId)
      .maybeSingle();

    if (existing) {
      return NextResponse.json({ received: true });
    }

    await supabaseAdmin.from("stripe_events").insert({
      id: eventId,
      type: event.type,
      created_at: new Date().toISOString(),
    });

    /* =========================
       CHECKOUT SUCCESS
    ========================= */
    if (event.type === "checkout.session.completed") {
      const session = event.data.object as Stripe.Checkout.Session;

      const type = session.metadata?.type as MetaType;

      /* =========================
         ESCROW FLOW
      ========================= */
      if (type === "escrow") {
        const escrowId = session.metadata?.escrowId;

        if (!escrowId) {
          console.error("Missing escrowId");
          return NextResponse.json({ received: true });
        }

        await escrowEngine("FUND", escrowId);

        console.log("💰 ESCROW RELEASED:", escrowId);
      }

      /* =========================
         AGREEMENT FLOW
      ========================= */
      if (type === "agreement") {
        const agreementId = session.metadata?.agreementId;

        if (!agreementId) {
          console.error("Missing agreementId");
          return NextResponse.json({ received: true });
        }

        await agreementEngine("ACCEPT", agreementId);

        console.log("📄 AGREEMENT ACCEPTED:", agreementId);
      }

      /* =========================
         BOOST FLOW
      ========================= */
      if (type === "boost") {
        const listingId = session.metadata?.listingId;

        if (!listingId) {
          console.error("Missing listingId");
          return NextResponse.json({ received: true });
        }

        await supabaseAdmin
          .from("listings")
          .update({
            boosted: true,
            boosted_at: new Date().toISOString(),
          })
          .eq("id", listingId);

        console.log("🚀 BOOST ACTIVATED:", listingId);
      }
    }

    /* =========================
       PAYMENT FAILED
    ========================= */
    if (event.type === "payment_intent.payment_failed") {
      console.log("❌ Payment failed:", event.id);
    }

    /* =========================
       REFUND
    ========================= */
    if (event.type === "charge.refunded") {
      const charge = event.data.object as Stripe.Charge;
      console.log("🔁 Refund:", charge.id);
    }

    return NextResponse.json({ received: true });
  } catch (err: any) {
    console.error("Webhook error:", err.message);

    return NextResponse.json(
      { error: "Webhook failed" },
      { status: 500 }
    );
  }
}

import { stripe } from "@/lib/infrastructure/stripe/client";
import { NextResponse } from "next/server";
import {
  isEventProcessed,
  markEventProcessed,
} from "@/lib/payments/webhookGuard";
import { escrowEngine } from "@/lib/domain/escrow/escrowEngine";

export async function handleStripeWebhook(req: Request) {
  const body = await req.text();
  const signature = req.headers.get("stripe-signature")!;
  const secret = process.env.STRIPE_WEBHOOK_SECRET!;

  let event;

  /* =========================
     1. VERIFY STRIPE SIGNATURE
  ========================= */
  try {
    event = stripe.webhooks.constructEvent(body, signature, secret);
  } catch (err) {
    return NextResponse.json(
      { error: "Invalid signature" },
      { status: 400 }
    );
  }

  const eventId = event.id;

  /* =========================
     2. IDEMPOTENCY CHECK
  ========================= */
  if (await isEventProcessed(eventId)) {
    return NextResponse.json({ ok: true, skipped: true });
  }

  /* =========================
     3. ONLY HANDLE SUCCESS PAYMENTS
  ========================= */
  if (event.type !== "checkout.session.completed") {
    return NextResponse.json({ ok: true, ignored: true });
  }

  const session = event.data.object as {
    metadata?: {
      escrowId?: string;
    };
  };

  const escrowId = session.metadata?.escrowId;

  if (!escrowId) {
    return NextResponse.json(
      { error: "Missing escrowId" },
      { status: 400 }
    );
  }

  /* =========================
     4. LOCK EVENT (BEFORE BUSINESS LOGIC)
  ========================= */
  await markEventProcessed({
    id: eventId,
    provider: "stripe",
    escrowId,
    raw: event,
  });

  /* =========================
     5. BUSINESS LOGIC (ESCROW FUND)
  ========================= */
  await escrowEngine("FUND", escrowId);

  return NextResponse.json({ ok: true });
}

      `https://api.chapa.co/v1/transaction/verify/${input.tx_ref}`,

      data?.status === "success" &&
      data?.data?.status === "success";

      throw new Error("Payment not confirmed");
    }

      data,
    };

     CREATE PAYMENT INTENT
  ------------------------- */
  async createIntent(input: CreatePaymentIntentInput) {
    const supabase = await createServerSupabase();

    // 1. Create internal payment record first
    const { data: payment, error } = await supabase
      .from("payments")
      .insert({
        campaign_id: input.campaignId,
        provider: input.provider,
        status: "pending",
        amount: 0,
      })
      .select()
      .single();

    if (error) {
      throw new Error(error.message);
    }

    // 2. Simulate external provider response
    const externalId =
      input.provider === "stripe"
        ? `stripe_${Date.now()}`
        : `chapa_${Date.now()}`;

    // 3. Update payment with provider reference
    await supabase
      .from("payments")
      .update({
        provider_ref: externalId,
      })
      .eq("id", payment.id);

    return {
      paymentId: payment.id,
      provider: input.provider,
      providerRef: externalId,
      status: "pending",
    };
  },

  /* -------------------------
     VERIFY PAYMENT
  ------------------------- */
  async verifyPayment(paymentId: string) {
    const supabase = await createServerSupabase();

    const { data: payment, error } = await supabase
      .from("payments")
      .update({
        status: "paid",
        paid_at: new Date().toISOString(),
      })
      .eq("id", paymentId)
      .select()
      .single();

    if (error) {
      throw new Error(error.message);
    }
